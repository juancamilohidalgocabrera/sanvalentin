  <script>
    const noBtn = document.getElementById("noBtn");
    const yesBtn = document.getElementById("yesBtn");
    const messageFinal = document.getElementById("messageFinal");
    const counter = document.getElementById("counter");
    const imgLove = document.getElementById("imgLove");

    let noCount = 0;
    let yesScale = 1;

    const imagesNo = [
      "images/perrito.png",
      "images/ardillas.png",
      "images/ardillas_beso.png",
      "images/gatos.png",
      "images/pinguino.png"
    ];

    noBtn.addEventListener("mouseover", () => {
      if (noCount >= 5) return;

      imgLove.src = imagesNo[noCount];

      noCount++;
      counter.textContent = "Intentos de NO: " + noCount + " / 5";

      let x = Math.random() * 250 - 120;
      let y = Math.random() * 180 - 90;
      noBtn.style.transform = "translate(" + x + "px, " + y + "px)";

      yesScale += 0.35;
      yesBtn.style.transform = "scale(" + yesScale + ")";

      if (noCount >= 5) {
        noBtn.disabled = true;
        noBtn.style.background = "#999";
        noBtn.style.cursor = "not-allowed";
        noBtn.textContent = "Ya no 😏";
        counter.textContent = "Ya no puedes decir que no 😌💖";

        yesBtn.style.transform = "scale(3)";
      }
    });

    yesBtn.addEventListener("click", () => {
      imgLove.src = "images/nosotros.jpg";
      messageFinal.style.display = "block";
      createHearts();
    });

    function createHearts() {
      for (let i = 0; i < 45; i++) {
        const heart = document.createElement("div");
        heart.classList.add("heart");
        heart.textContent = "💖";

        heart.style.left = (Math.random() * window.innerWidth) + "px";
        heart.style.fontSize = (Math.random() * 25 + 15) + "px";
        heart.style.animationDuration = (Math.random() * 2 + 3) + "s";

        document.body.appendChild(heart);

        setTimeout(() => {
          heart.remove();
        }, 5000);
      }
    }
  </script>
